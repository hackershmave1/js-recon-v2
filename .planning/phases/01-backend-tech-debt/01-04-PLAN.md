```python
# AFTER:
import subprocess
import sys
import os

@app.on_event("startup")
def on_startup():
    # Run pending Alembic migrations instead of create_all.
    # alembic.ini is at the project root (one level above api/).
    project_root = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
    result = subprocess.run(
        ["alembic", "upgrade", "head"],
        cwd=project_root,
        capture_output=True,
        text=True,
    )
    if result.returncode != 0:
        raise RuntimeError(f"Alembic upgrade head failed:\n{result.stderr}")
```

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-01-04-02 | Denial of Service | alembic upgrade head on every startup | accept | Alembic is idempotent when already at head; overhead is negligible (metadata table check only) |
